====== BOT DEPLOY PLAN ======
project: w4p_get_seated_navigation_bot

objective: >
  Build a minimal get-seated automation layer only.
  The bot gets one assigned player to the correct PokerBet table,
  unticks "Wait for BB", performs the required initial CHECK/CALL
  if needed to complete seating, then stops. All further action is
  controlled manually through /remote/.

core_rule: >
  This is NOT a gameplay bot. It is only a seating/bootstrap bot.

architecture:
  instance_model:
	rule: "1 EC2 instance = 1 player session"
	browser: "Chromium preferred"
	automation: "Playwright preferred, Selenium acceptable"
	os: "Ubuntu EC2"
	db: "MySQL"
	remote_control: "/remote/"
	snapshot_bridge: "existing w4p.js"
	engine_stream: "unchanged"

  flow:
	- "Read assigned player/table from MySQL"
	- "Launch browser"
	- "Login if needed"
	- "Open assigned table"
	- "Untick Wait for BB"
	- "Click CHECK/CALL only if required for seating"
	- "Confirm snapshot stream is live"
	- "Stop automation"
	- "Leave all further actions to /remote/"

strict_boundaries:
  allowed_clicks:
	- "login"
	- "table open"
	- "seat selection"
	- "buy-in confirmation"
	- "untick Wait for BB"
	- "initial CHECK/CALL only for get-seated completion"

  forbidden_clicks:
	- "raise"
	- "all-in"
	- "cashout"
	- "fold"
	- "any repeated gameplay action"
	- "any decision-based betting action"

  after_seated:
	rule: "Automation must stop"
	control_owner: "/remote/"
	action_path: "/remote/ -> command queue -> w4p.js -> native PokerBet DOM click"

remote_ui_rules:
  render_only:
	- "heroes/self players with their own live snapshot"
	- "players with valid seat_token"
	- "players whose own bot/session sent the snapshot"

  never_render:
	- "villains"
	- "face-down hands"
	- "empty seats"
	- "placeholder seats"
	- "inferred seats from another player's snapshot"

  bottom_bar:
	keep: true
	purpose: "emergency fallback control"
	style: "sleek/minimized, not bulky"
	behavior:
	  - "shows buttons only for active hero"
	  - "collapses/disables when no active hero"
	  - "mirrors available_actions from PokerBet DOM"

engine_rules:
  must_remain_unchanged: true
  do_not_touch:
	- "/engine/"
	- "engine payload format"
	- "collector endpoints"
	- "existing command queue"
	- "existing w4p.js command bridge unless required for seating validation"

mysql_assignment_fields:
  required:
	- "player_name"
	- "username"
	- "password_ref"
	- "target_table"
	- "target_variant"
	- "target_stakes"
	- "buyin_mode"
	- "status"
	- "ec2_instance_id"
	- "last_heartbeat_ts"
	- "last_snapshot_ts"
	- "last_error"

statuses:
  - "idle"
  - "launching"
  - "logging_in"
  - "opening_table"
  - "seating"
  - "unticking_wait_for_bb"
  - "initial_check_call"
  - "snapshot_wait"
  - "snapshot_live"
  - "stopped"
  - "failed"

validation_required:
  seating:
	- "Player reaches correct table"
	- "Wait for BB is unticked"
	- "Initial CHECK/CALL executed only if required"
	- "Automation stops after seating"

  snapshot:
	- "/api/collector/latest updates"
	- "/api/table/latest shows player"
	- "seat_token exists"
	- "active player state passes through if available"

  remote:
	- "Only hero/self panels render"
	- "Villains do not render at all"
	- "Bottom emergency bar remains"
	- "Remote click still triggers native PokerBet DOM action"

  engine:
	- "/engine/ stream unchanged"
	- "No collector errors"
	- "No payload-breaking changes"

logging:
  required:
	- "current status"
	- "current URL"
	- "selected player"
	- "selected table"
	- "Wait for BB unticked result"
	- "initial CHECK/CALL result"
	- "snapshot live confirmation"
	- "screenshots on failure"
	- "browser console errors"
	- "last heartbeat"

failure_rules:
  table_not_found: "mark failed, save screenshot"
  login_failed: "mark failed, save screenshot"
  wait_for_bb_not_found: "log warning, continue only if seating can proceed"
  check_call_not_available: "log info, continue to snapshot wait"
  no_snapshot_after_30s: "restart browser once"
  no_snapshot_after_60s: "mark failed/stale"

agent_instruction: >
  Build only the get-seated bootstrap flow.
  The first action after opening the table is to untick Wait for BB.
  The second action is CHECK/CALL only if needed to complete seating.
  Then stop. Do not automate gameplay. Do not touch /engine/.
  Do not redesign /remote/. Do not render villains. Keep the bottom
  emergency bar. All gameplay actions after seating must come from /remote/.

success_definition: >
  One EC2 instance logs in one player, opens the assigned table,
  unticks Wait for BB, completes seating with CHECK/CALL if required,
  confirms snapshot live, then stops automation while /remote/ takes over.
